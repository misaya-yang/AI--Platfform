  -e POSTGRES_PASSWORD=111111 ^
  -p 5432:5432 ^
  -v pg-data:/var/lib/postgresql ^
  --restart unless-stopped ^
  postgres:latest

  -p 6379:6379 ^
  -v redis-data:/data ^
  --restart unless-stopped ^
  redis:latest redis-server --appendonly yes --requirepass 111111
